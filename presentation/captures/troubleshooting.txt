# Exercise 2: a Service whose selector matches no pods
$ kubectl apply -f k8s/exercises/02-wrong-service-selector.yaml
deployment.apps/lab-api-gateway created
service/lab-api-gateway created

$ kubectl -n crm-lab run probe --rm -i --image=busybox:1.37 -- wget -qO- -T 3 http://lab-api-gateway/healthz
wget: can't connect to remote host (172.20.225.80): Connection refused

$ kubectl -n crm-lab get endpointslices -l kubernetes.io/service-name=lab-api-gateway
NAME                    ADDRESSTYPE   PORTS     ENDPOINTS   AGE
lab-api-gateway-k4sfm   IPv4          <unset>   <unset>     11s

@@ 3
$ kubectl -n crm-lab describe svc lab-api-gateway | grep Selector

$ kubectl -n crm-lab get pods -l app=lab-api-gateway --show-labels
NAME STATUS LABELS
lab-api-gateway-5c77c4df4-482tq Running app=lab-api-gateway,tier=edge
lab-api-gateway-5c77c4df4-v5n69 Running app=lab-api-gateway,tier=edge
